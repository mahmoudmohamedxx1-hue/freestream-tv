     1→// Playlist catalog — organized by PLATFORM
     2→// Each platform (Pluto TV, Samsung TV Plus, LG Channels, Tubi, Roku, etc.)
     3→// is a Provider with sub-playlists grouped under categories like
     4→// "By Country" or "All Channels". Channel-type filtering (Sports/News/Movies/etc.)
     5→// happens in the sidebar via the M3U group-title field, not at the provider level.
     6→
     7→export type PlaylistItem = {
     8→  id: string
     9→  name: string
    10→  flag?: string
    11→  url: string
    12→  /** Channel count hint shown in UI */
    13→  count?: number
    14→}
    15→
    16→export type ProviderCategory = {
    17→  id: string
    18→  name: string
    19→  flag?: string
    20→  /** If set, clicking this category loads this URL directly (no sub-playlist picker) */
    21→  directUrl?: string
    22→  /** If set, this category contains multiple sub-playlists to pick from */
    23→  playlists?: PlaylistItem[]
    24→}

   813→export function getProviderById(id: string): Provider | undefined {
   814→  return PROVIDERS.find((p) => p.id === id)
   815→}
   816→
   817→export function getCategoryById(providerId: string, categoryId: string): ProviderCategory | undefined {
   818→  return getProviderById(providerId)?.categories.find((c) => c.id === categoryId)
   819→}
   820→
   821→export function getPlaylistById(providerId: string, categoryId: string, playlistId: string): PlaylistItem | undefined {
   822→  return getCategoryById(providerId, categoryId)?.playlists?.find((p) => p.id === playlistId)
   823→}
   824→
   825→/** Resolve a provider+category+playlist path to a URL */
   826→export function resolvePlaylistUrl(providerId: string, categoryId: string, playlistId?: string): string | undefined {
   827→  const cat = getCategoryById(providerId, categoryId)
   828→  if (!cat) return undefined
   829→  if (cat.directUrl) return cat.directUrl
   830→  if (playlistId && cat.playlists) {
   831→    const pl = cat.playlists.find((p) => p.id === playlistId)
   832→    return pl?.url
   833→  }
   834→  if (cat.playlists && cat.playlists.length > 0) return cat.playlists[0].url
   835→  return undefined
   836→}
